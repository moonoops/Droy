cmake_minimum_required(VERSION 3.22)
project(droy_embedding LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Droy embedding targets Android arm64-v8a")
endif()
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${DROY_SENTENCEPIECE_SOURCE}" "${CMAKE_BINARY_DIR}/sentencepiece")
add_library(droy_existing_ort SHARED IMPORTED)
set_target_properties(droy_existing_ort PROPERTIES IMPORTED_LOCATION "${DROY_ORT_LIBRARY}")
add_library(droy_embedding SHARED embedding_jni.cpp)
target_include_directories(droy_embedding PRIVATE "${DROY_ORT_HEADERS}" "${DROY_SENTENCEPIECE_SOURCE}" "${DROY_SENTENCEPIECE_SOURCE}/src")
target_link_libraries(droy_embedding PRIVATE sentencepiece-static droy_existing_ort log)
target_compile_options(droy_embedding PRIVATE -O2 -fvisibility=hidden)
target_link_options(droy_embedding PRIVATE "-Wl,-z,max-page-size=16384")
