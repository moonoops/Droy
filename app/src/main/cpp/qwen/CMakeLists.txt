cmake_minimum_required(VERSION 3.22)
project(droy_qwen_tts LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT ANDROID OR NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Droy Qwen TTS targets Android arm64-v8a")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${DROY_GGML_SOURCE}" "${CMAKE_BINARY_DIR}/ggml")

# The pinned upstream root adds desktop -march=native and builds CLI/tests.
# Compile its library sources directly for the Android baseline CPU instead.
file(GLOB_RECURSE QWEN_SOURCES CONFIGURE_DEPENDS "${DROY_QWEN_SOURCE}/src/*.cpp")
list(FILTER QWEN_SOURCES EXCLUDE REGEX "/(main|qwen3_tts_jni|qwen3_tts_quantize)\\.cpp$")
add_library(droy_qwen_tts SHARED qwen_jni.cpp ${QWEN_SOURCES})
target_include_directories(droy_qwen_tts PRIVATE "${DROY_QWEN_SOURCE}/src" "${DROY_GGML_SOURCE}/include")
target_link_libraries(droy_qwen_tts PRIVATE ggml ggml-base ggml-cpu log)
target_compile_options(droy_qwen_tts PRIVATE -O3 -fvisibility=hidden)
target_link_options(droy_qwen_tts PRIVATE "-Wl,-z,max-page-size=16384")
